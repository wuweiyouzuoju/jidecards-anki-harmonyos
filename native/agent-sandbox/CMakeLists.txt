cmake_minimum_required(VERSION 3.5.0)
project(jidecards_agent_sandbox LANGUAGES C CXX)
if(OHOS_ARCH STREQUAL "arm64-v8a")
    set(SANDBOX_TARGET "aarch64-unknown-linux-ohos")
elseif(OHOS_ARCH STREQUAL "x86_64")
    set(SANDBOX_TARGET "x86_64-unknown-linux-ohos")
else()
    message(FATAL_ERROR "Unsupported sandbox architecture: ${OHOS_ARCH}")
endif()
set(SANDBOX_ARCHIVE "${CMAKE_CURRENT_SOURCE_DIR}/target/${SANDBOX_TARGET}/release/libjidecards_agent_sandbox.a")
if(NOT EXISTS "${SANDBOX_ARCHIVE}")
    message(FATAL_ERROR "Missing sandbox archive; run tools/build-agent-sandbox.ps1 -Target app")
endif()
add_library(agent_sandbox SHARED napi.cpp)
set_target_properties(agent_sandbox PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/sandbox_out")
target_include_directories(agent_sandbox PRIVATE include)
target_link_libraries(agent_sandbox PRIVATE "${SANDBOX_ARCHIVE}" libace_napi.z.so)
target_link_options(agent_sandbox PRIVATE "-Wl,--exclude-libs,ALL")
